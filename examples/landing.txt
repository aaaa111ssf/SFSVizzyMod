# ===== SFS Vizzy landing example (best on the Moon / airless bodies) =====
# How to run: press RUN once you are on the descent. Make sure the engine
# is lit and the RCS has fuel.
# With an atmosphere: add a parachute block first, wait for the speed to
# drop, then run this program.
# Note: parameters must live inside on start (outside it would be treated
# as another parallel entry point).

on start
  # ---- tunables ----
  set vtouch = 1.2       # target touchdown speed, m/s
  set profile = 0.55     # descent curve: 0.4 gentle / 0.7 aggressive
  set gain = 0.06        # throttle gain: lower if it shakes, raise if sluggish
  set maxdesc = 60       # max descent rate, m/s

  rcs on
  throttle on

  # 0) thrust check
  wait 0.3
  if MaxTWR < 1.05
    fail "MaxTWR < 1.05: thrust too low to land"
  end

  # 1) kill horizontal speed: burn against the velocity vector
  if HorizontalSpeed > 3
    turn to retrograde in 2
    set throttle to 1
    while HorizontalSpeed > 3
      wait 0.2
    end
  end

  # 2) vertical descent: the target descent rate comes from the remaining
  #    height and the available deceleration, then a proportional throttle
  #    tracks it
  turn to up in 2
  while AltitudeAGL > 1.5
    set amax = max(0.5, MaxTWR * 9.8)                 # max acceleration, m/s^2
    set v = 0 - VerticalSpeed                         # descent as positive
    set h = max(AltitudeAGL, 0.5)

    # gentle curve: descend at 0.55 of the speed that could still stop in time
    set vt = clamp(sqrt(2 * max(0.5, amax - Gravity) * h) * profile, vtouch, maxdesc)

    # force a slow rate over the last 5 m so the touchdown is soft
    if h < 5
      set vt = vtouch
    end

    # hover throttle plus a speed error correction
    set throttle to clamp(Gravity / amax + (v - vt) * gain, 0, 1)
    wait 0.1
  end

  # 3) touchdown
  throttle off
  rcs off
  log "Touchdown: {VerticalSpeed} m/s at {AltitudeAGL} m AGL"
end
