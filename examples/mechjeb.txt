# ===== SFS Vizzy autopilot collection (MechJeb style, pick a mode) =====
# How to run: change mode in on start, then press RUN.
#   mode 1 launch to orbit   2 circularize   3 change apsides
#   mode 4 auto landing      5 parachute landing   6 target navigation
#
# Apply with Ctrl+Enter in the code view, then hit run.
# Note: expressions have no and / or. Use nested if plus break instead
# (this file already does it that way).

on start
  set mode = 2             # which function to run, see above
  set targetAp = 100000    # mode 1/3: target apoapsis altitude (m)
  set targetPe = 100000    # mode 3: target periapsis altitude (m)
  set tol = 3000           # circularize / apsides tolerance (m)

  # ================= 0 do nothing =================
  if mode == 0
    throttle off
    log "mj0: nothing to do"
  end

  # ================= 1 launch to orbit =================
  if mode == 1
    log "mj1: liftoff"
    throttle on
    set throttle to 1
    wait until AltitudeAGL > 300

    # pitch over as altitude grows, from straight up towards horizontal
    # SFS angle: 0 = straight up, 90 = horizontal. frac 0->1 gives 0->85
    while Apoapsis < targetAp
      set frac = clamp(AltitudeAGL / 45000, 0, 1)
      turn to (85 * frac) in 0.5
      set throttle to 1
      wait 0.5
    end
    log "mj1: apoapsis = {Apoapsis}"

    # fly level for a while so the periapsis climbs too
    turn to 90 in 3
    set throttle to 0.25
    wait 8
    log "mj1: coasting ap={Apoapsis} pe={Periapsis}"

    # circularize
    circularize tol
    log "mj1: orbit done ap={Apoapsis} pe={Periapsis}"
  end

  # ================= 2 circularize =================
  if mode == 2
    # refuse to burn on the pad: there is no orbit to circularize yet
    if Apoapsis < 2000
      fail "mj2: not in orbit yet, launch first"
    end
    log "mj2: circularize, start ap={Apoapsis} pe={Periapsis}"
    circularize tol
    log "mj2: done ap={Apoapsis} pe={Periapsis}"
  end

  # ================= 3 change apsides =================
  # no and operator in expressions, so use while + if + break
  # to mean "not there yet AND still have tries left"
  if mode == 3
    if Apoapsis < 2000
      fail "mj3: not in orbit yet, launch first"
    end
    rcs on
    set tries = 0
    set maxTries = 12

    while tries < maxTries
      if abs(Apoapsis - targetAp) <= tol
        break
      end

      # wait until we are near an apsis before burning
      wait until abs(VerticalSpeed) < 3

      if Altitude < (Apoapsis + Periapsis) / 2
        # near periapsis: change the apoapsis
        if targetAp > Apoapsis
          turn to prograde in 1
        else
          turn to retrograde in 1
        end
      else
        # near apoapsis: change the periapsis
        if targetPe > Periapsis
          turn to prograde in 1
        else
          turn to retrograde in 1
        end
      end

      set throttle to 0.9
      wait 2
      throttle off

      set tries = tries + 1
      wait 15
    end
    log "mj3: apsides ap={Apoapsis} pe={Periapsis} tries={tries}"
  end

  # ================= 4 auto landing (with thrust) =================
  if mode == 4
    deploy all
    wait 1
    log "mj4: legs out, alt={AltitudeAGL} vs={VerticalSpeed}"

    # kill horizontal speed first
    if HorizontalSpeed > 5
      turn to retrograde in 2
      set throttle to 1
      wait until HorizontalSpeed < 5
      log "mj4: horizontal speed killed"
    end

    # step down: fast up high, slow near the ground
    # negatives go into variables, because the hold target field
    # only takes one token without spaces
    turn to up in 2
    set vs1 = 0 - 20
    set vs2 = 0 - 8
    set vs3 = 0 - 3
    set vs4 = 0 - 1

    hold vspeed vs1 5
    hold vspeed vs2 5
    hold vspeed vs3 6
    hold vspeed vs4 10

    set throttle to 0
    log "mj4: touchdown vs={VerticalSpeed} agl={AltitudeAGL}"
  end

  # ================= 5 parachute landing (with atmosphere) =================
  if mode == 5
    deploy all
    parachute all deploy
    log "mj5: chute out, vs={VerticalSpeed}"

    # once the chute slows us down, add a little thrust for a soft touchdown
    set vsSoft = 0 - 4
    hold vspeed vsSoft 20
    set throttle to 0
    log "mj5: landed vs={VerticalSpeed} agl={AltitudeAGL}"
  end

  # ================= 6 target navigation =================
  # select a target in the map first, otherwise it fails and shows a red bar
  if mode == 6
    log "mj6: target dist={TargetDistance} m, bearing={TargetBearing} deg"
    turn to target
    log "mj6: pointed at target"

    set n = 0
    while n < 3
      wait 5
      turn to target
      set n = n + 1
      log "mj6: track {n}/3, dist={TargetDistance}"
    end
    log "mj6: done"
  end
end
